    ```bash

      xargs -I{} sh -c 'grep -l "/Users/aslaii" /Users/aslaii/dotfiles/{} 2>/dev/null' | \

    echo "Files to update: $(echo "$TRACKED_FILES" | wc -l)"

    # Run the replacement on each file
    ```
